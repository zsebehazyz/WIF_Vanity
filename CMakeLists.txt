cmake_minimum_required(VERSION 3.24)
project(VanityWIF VERSION 1.0.0 LANGUAGES CXX)
option(VANITY_CUDA "Build real NVIDIA CUDA backend (requires CUDA Toolkit)" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Threads REQUIRED)
add_library(vanity_core src/codec.cpp src/verify.cpp src/search.cpp src/state.cpp)
target_include_directories(vanity_core PUBLIC include)
target_link_libraries(vanity_core PUBLIC Threads::Threads)
if(WIN32)
  target_compile_definitions(vanity_core PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(vanity_core PUBLIC bcrypt)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(vanity_core PUBLIC OpenSSL::Crypto)
endif()
set(VANITY_CUDA_FATBIN "" CACHE FILEPATH "Use a precompiled CUDA fatbin (allows Windows cross compilation)")
set(VANITY_CUDA_ARCH "86" CACHE STRING "Minimum CUDA compute capability, RTX 3050 = 86")
if(VANITY_CUDA)
  if(NOT VANITY_CUDA_FATBIN)
    set(CMAKE_CUDA_ARCHITECTURES ${VANITY_CUDA_ARCH})
    enable_language(CUDA)
    set(VANITY_CUDA_FATBIN "${CMAKE_CURRENT_BINARY_DIR}/vanity.fatbin")
    add_custom_command(OUTPUT "${VANITY_CUDA_FATBIN}"
      COMMAND "${CMAKE_CUDA_COMPILER}" --fatbin --std=c++17 -O3
        "-gencode=arch=compute_${VANITY_CUDA_ARCH},code=[sm_${VANITY_CUDA_ARCH},compute_${VANITY_CUDA_ARCH}]"
        "-I${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/src/cuda_kernel.cu"
        -o "${VANITY_CUDA_FATBIN}"
      DEPENDS src/cuda_kernel.cu include/vanity/sha256.hpp VERBATIM)
  endif()
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/cuda_image.hpp"
    COMMAND "${CMAKE_COMMAND}" "-DINPUT=${VANITY_CUDA_FATBIN}"
      "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/cuda_image.hpp" "-DARCH=${VANITY_CUDA_ARCH}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed.cmake"
    DEPENDS "${VANITY_CUDA_FATBIN}" cmake/embed.cmake VERBATIM)
  target_sources(vanity_core PRIVATE src/cuda_driver.cpp "${CMAKE_CURRENT_BINARY_DIR}/cuda_image.hpp")
  target_include_directories(vanity_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(vanity_core PRIVATE ${CMAKE_DL_LIBS})
  target_compile_definitions(vanity_core PUBLIC VANITY_HAS_CUDA=1)
else()
  target_sources(vanity_core PRIVATE src/cuda_stub.cpp)
endif()
if(MSVC)
  target_compile_options(vanity_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/O2>)
else()
  target_compile_options(vanity_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
add_executable(vanity-cli src/cli.cpp)
target_link_libraries(vanity-cli PRIVATE vanity_core)
if(WIN32)
  enable_language(RC)
  add_executable(VanityWIF WIN32 src/win_gui.cpp src/app.rc)
  target_include_directories(VanityWIF PRIVATE src)
  if(MSVC)
    target_compile_options(VanityWIF PRIVATE /utf-8 /W4)
    target_link_options(VanityWIF PRIVATE /MANIFEST:NO)
  endif()
  target_compile_definitions(VanityWIF PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(VanityWIF PRIVATE vanity_core comctl32 shell32 ole32)
  if(MINGW)
    target_link_options(VanityWIF PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(vanity-cli PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(vanity-tests tests/test_main.cpp)
  target_link_libraries(vanity-tests PRIVATE vanity_core)
  if(MINGW)
    target_link_options(vanity-tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME core COMMAND vanity-tests)
  set_tests_properties(core PROPERTIES TIMEOUT 120)
  if(WIN32)
    add_executable(vanity-gui-smoke tests/gui_smoke.cpp)
    target_link_libraries(vanity-gui-smoke PRIVATE vanity_core user32)
    target_compile_definitions(vanity-gui-smoke PRIVATE UNICODE _UNICODE NOMINMAX)
    if(MINGW)
      target_link_options(vanity-gui-smoke PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    option(VANITY_GUI_TESTS "Run interactive desktop GUI smoke test through CTest" OFF)
    if(VANITY_GUI_TESTS)
      add_test(NAME gui COMMAND vanity-gui-smoke)
      set_tests_properties(gui PROPERTIES TIMEOUT 120)
    endif()
  endif()
endif()
