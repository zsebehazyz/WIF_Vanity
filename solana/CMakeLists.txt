cmake_minimum_required(VERSION 3.24)
project(SolanaVanity VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(SOLVAN_CUDA "Build real CUDA Ed25519 backend" OFF)
set(SOLVAN_FATBIN "" CACHE FILEPATH "Existing GPU fatbin for Windows cross compilation")
set(SOLVAN_ARCH 86 CACHE STRING "Minimum NVIDIA compute capability")
find_package(Threads REQUIRED)
add_library(solvan_crypto STATIC third_party/ed25519/keypair.c third_party/ed25519/fe.c third_party/ed25519/ge.c third_party/ed25519/sha512.c)
target_include_directories(solvan_crypto PUBLIC third_party/ed25519)
add_library(solvan_verify STATIC third_party/monocypher/monocypher.c third_party/monocypher/monocypher-ed25519.c)
target_include_directories(solvan_verify PUBLIC third_party/monocypher)
add_library(solvan_core STATIC src/crypto.cpp src/verify.cpp src/prefix.cpp src/storage.cpp src/search.cpp)
target_include_directories(solvan_core PUBLIC include)
target_link_libraries(solvan_core PUBLIC solvan_crypto solvan_verify Threads::Threads)
if(WIN32)
  target_compile_definitions(solvan_core PUBLIC _WIN32_WINNT=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(solvan_core PUBLIC bcrypt crypt32)
endif()
if(SOLVAN_CUDA)
  if(NOT SOLVAN_FATBIN)
    set(SOLVAN_PORT_TESTS ON)
    set(CMAKE_CUDA_ARCHITECTURES ${SOLVAN_ARCH})
    enable_language(CUDA)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    file(GLOB SOLVAN_REF_SOURCES CONFIGURE_DEPENDS third_party/ed25519/*)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/cuda-crypto.stamp"
      COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_cuda.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ed25519" "${CMAKE_CURRENT_BINARY_DIR}/cuda-crypto"
      COMMAND ${CMAKE_COMMAND} -E touch "${CMAKE_CURRENT_BINARY_DIR}/cuda-crypto.stamp"
      DEPENDS ${SOLVAN_REF_SOURCES} tools/prepare_cuda.py VERBATIM)
    set(SOLVAN_FATBIN "${CMAKE_CURRENT_BINARY_DIR}/solana.fatbin")
    add_custom_command(OUTPUT "${SOLVAN_FATBIN}"
      COMMAND ${CMAKE_CUDA_COMPILER} --fatbin --std=c++17 -O3 -Xptxas=-v
        "-gencode=arch=compute_${SOLVAN_ARCH},code=[sm_${SOLVAN_ARCH},compute_${SOLVAN_ARCH}]"
        "-I${CMAKE_CURRENT_BINARY_DIR}/cuda-crypto" "-I${CMAKE_CURRENT_SOURCE_DIR}/include"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/kernel.cu" -o "${SOLVAN_FATBIN}"
      DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/cuda-crypto.stamp" src/kernel.cu include/solvan/device.hpp include/solvan/device_crypto.hpp VERBATIM)
  endif()
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/cuda_image.hpp"
    COMMAND ${CMAKE_COMMAND} "-DINPUT=${SOLVAN_FATBIN}" "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/cuda_image.hpp"
      "-DARCH=${SOLVAN_ARCH}" -P "${CMAKE_CURRENT_SOURCE_DIR}/../cmake/embed.cmake"
    DEPENDS "${SOLVAN_FATBIN}" "${CMAKE_CURRENT_SOURCE_DIR}/../cmake/embed.cmake" VERBATIM)
  target_sources(solvan_core PRIVATE src/cuda_driver.cpp "${CMAKE_CURRENT_BINARY_DIR}/cuda_image.hpp")
  target_include_directories(solvan_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(solvan_core PRIVATE ${CMAKE_DL_LIBS})
else()
  target_sources(solvan_core PRIVATE src/cuda_stub.cpp)
endif()
if(MSVC)
  target_compile_options(solvan_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(solvan_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(solana-cli src/cli.cpp)
target_link_libraries(solana-cli PRIVATE solvan_core)
if(WIN32)
  enable_language(RC)
  add_executable(SolanaVanity WIN32 src/win_gui.cpp src/app.rc)
  target_compile_definitions(SolanaVanity PRIVATE UNICODE _UNICODE)
  target_include_directories(SolanaVanity PRIVATE src)
  target_link_libraries(SolanaVanity PRIVATE solvan_core comctl32 shell32 ole32)
  if(MINGW)
    target_link_options(SolanaVanity PRIVATE -municode)
  else()
    target_compile_options(SolanaVanity PRIVATE /utf-8 /W4)
    target_link_options(SolanaVanity PRIVATE /MANIFEST:NO)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(solana-tests tests/test_main.cpp)
  target_link_libraries(solana-tests PRIVATE solvan_core)
  if(SOLVAN_PORT_TESTS)
    target_sources(solana-tests PRIVATE tests/ported_crypto.cpp "${CMAKE_CURRENT_BINARY_DIR}/cuda-crypto.stamp")
    target_include_directories(solana-tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/cuda-crypto")
    target_compile_definitions(solana-tests PRIVATE SOLVAN_PORT_HOST=1)
  endif()
  add_test(NAME solana-core COMMAND solana-tests)
  set_tests_properties(solana-core PROPERTIES TIMEOUT 180)
  if(WIN32)
    add_executable(solana-gui-smoke tests/gui_smoke.cpp)
    target_compile_definitions(solana-gui-smoke PRIVATE UNICODE _UNICODE)
    target_link_libraries(solana-gui-smoke PRIVATE solvan_core user32)
  endif()
endif()
if(MINGW)
  foreach(target SolanaVanity solana-cli solana-tests solana-gui-smoke)
    if(TARGET ${target})
      target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  endforeach()
endif()
